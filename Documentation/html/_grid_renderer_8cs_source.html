100 void Redraw()
101 {
102 foreach (var go in spawned)
103 Destroy(go);
104 spawned.Clear();
105
106 float half = (grid.size - 1) / 2f;
107 originOffset.localPosition = new Vector3(-half, -half, -half);
108
109
110 for (int x = 0; x < grid.size; x++)
111 for (int y = 0; y < grid.size; y++)
112 for (int z = 0; z < grid.size; z++)
113 {
114 var voxelData = grid.Get(x, y, z);
115 if (voxelData.IsVoid()) continue;
116
117 GameObject prefab = GetPrefab(voxelData.meshId);

120 voxelRender.transform.localPosition = new Vector3(x, y, z);
121 voxelRender.transform.localRotation = voxelData.GetOrientation();
122 voxelRender.transform.localScale = Vector3.Scale(voxelData.GetScale(), prefabScale);
123 voxelRender.GetComponent<Renderer>().material.color = GetColor(voxelData.meshId);
