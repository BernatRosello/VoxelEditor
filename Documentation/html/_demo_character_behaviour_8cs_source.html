104 private InteractionRequest GetRandomInteraction()
105 {
106 float totalWeight = 0f;
107
108 foreach (var entry in interactionWeights)
109 totalWeight += entry.weight;
110
111 if (totalWeight <= 0f)
112 throw new InvalidOperationException("No interaction requests have a positive weight.");
113
114 float random = Random.value * totalWeight;
115
116 foreach (var entry in interactionWeights)
117 {
118 random -= entry.weight;
