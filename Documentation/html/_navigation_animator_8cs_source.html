114 public void SetCurrentUp()
115 {
116 switch (navSettings.NavSurfMode)
117 {
118 case NavSurfaceMode.RaycastSurface:
119 // NOTE: NavMesh API sucks and there is currently NO WAY to get the normal of the NavMeshSurface at any given position.
120 // To work around this we can use a raycast, but it should be filtered only to include the collection of meshes that
121 // were used to bake the NavMesh in the first place (but even this is not a perfect solution since baking can severely
122 // change the Surface when comparing it to the base mesh).
123 if (Physics.Raycast(animatedTransform.position, -animatedTransform.up, out RaycastHit hit,
124 navSettings.SurfaceRayDistance,
125 navSettings.SurfaceMask))
126 {
127 surfaceUp = hit.normal;
128 }
129 break;
130 case NavSurfaceMode.SphereTransform:
131 surfaceUp = (animatedTransform.position - navSurfaceTransform.position).normalized;
132 break;
133 case NavSurfaceMode.FlatTransform:
134 surfaceUp = navSurfaceTransform.up;
135 break;
136 case NavSurfaceMode.WorldUp:
137 surfaceUp = Vector3.up;
138 break;
139 }

146 // To handle unexpected infinite values given out by the NavMesh plugin. As referenced by https://stackoverflow.com/a/67561314
147 public static float GetRemainingDistance(NavMeshAgent agent)
148 {
149 if (!Single.IsInfinity(agent.remainingDistance))
150 {
151 return agent.remainingDistance;
152 }
153
154 float distance = 0;
155 Vector3[] corners = agent.path.corners;
156
157 if (corners.Length > 2)
158 {
159 for (int i = 1; i < corners.Length; i++)
160 {
161 Vector2 previous = new Vector2(corners[i - 1].x, corners[i - 1].z);
162 Vector2 current = new Vector2(corners[i].x, corners[i].z);
163
164 distance += Vector2.Distance(previous, current);
165 }
166 }
167 else
168 {
169 distance = 0;
170 Debug.LogError("CAN'T FIND THE LENGTH OF THE REMAINING PATH OF THE AGENT!");
171 }
172
173 return distance;
174 }

176 private float GetSpeedAlongPath(float remainingDistance, float totalPathDistance)
177 {
178 if (totalPathDistance <= 0.01f)
179 return minimumSpeed;
180
181 float progress = 1f - Mathf.Clamp01(remainingDistance / totalPathDistance);
182
183 float normalizedSpeed;
184
185 string profileInfo;
186
187 if (totalPathDistance <= mediumPathLength)
188 {
189 float t = Mathf.InverseLerp(shortPathLength, mediumPathLength, totalPathDistance);
190
191 float shortValue = shortPathSpeedCurve.Evaluate(progress);
192
193 float mediumValue =
194 mediumPathSpeedCurve.Evaluate(progress);
195
196 normalizedSpeed =
197 Mathf.Lerp(
198 shortValue,
199 mediumValue,
200 t);
201
202 profileInfo =
203 $"SHORT→MEDIUM blend={t:F2} short={shortValue:F2} medium={mediumValue:F2}";
204 }
205 else
206 {
207 float t =
208 Mathf.InverseLerp(
209 mediumPathLength,
210 longPathLength,
211 totalPathDistance);
212
213 float mediumValue =
214 mediumPathSpeedCurve.Evaluate(progress);
215
216 float longValue =
217 longPathSpeedCurve.Evaluate(progress);
218
219 normalizedSpeed = Mathf.Lerp(mediumValue, longValue, t);
220
221 profileInfo =
222 $"MEDIUM→LONG blend={t:F2} medium={mediumValue:F2} long={longValue:F2}";
223 }
224
225 float finalSpeed =
226 Mathf.Lerp(
227 minimumSpeed,
228 maximumSpeed,
229 normalizedSpeed);
230
231 // Debug.Log(
232 // $"[PathSpeed] " +
233 // $"remaining={remainingDistance:F2}m " +
234 // $"total={totalPathDistance:F2}m " +
235 // $"progress={progress:P0} " +
236 // $"{profileInfo} " +
237 // $"normalized={normalizedSpeed:F2} " +
238 // $"speed={finalSpeed:F2}");
239
240 return finalSpeed;
241 }

291 private void ProcessMovement()
292 {
293 if (!navigationActive) return;
294
295 if (turnRequestActive)
296 {
297 ProcessTurnRequest();
298 return;
299 }
300
301 if (!agent.hasPath)
302 {
303 pathCachedFlag = false;
304 animator.SetBool("IsMoving", false);
305 animator.SetBool("IsTurning", false);
306 return;
307 }
308
309 if (!pathCachedFlag &&
310 !agent.pathPending)
311 {
312 cachedPathLength = GetRemainingDistance(agent);
313 pathCachedFlag = true;
314 }
315
316 float remainingDistance = GetRemainingDistance(agent);
317 smoothedSteeringTarget = agent.steeringTarget;
318 Vector3 toTarget = Vector3.ProjectOnPlane(smoothedSteeringTarget - animatedTransform.position, surfaceUp);
319
320 if (toTarget.sqrMagnitude < 0.001f)
321 return;
322
323 Vector3 desiredForward = toTarget.normalized;
324
325 float angleToTarget = Vector3.SignedAngle(animatedTransform.forward, desiredForward, surfaceUp);
326 bool facingTarget = IsFacingDirection(desiredForward);
327
328 bool isMoving = animator.GetBool("IsMoving");
329 bool isTurning = animator.GetBool("IsTurning");
330
331 //
332 // PRE-MOVEMENT PLANNING
333 //
334 if (!isMoving && !isTurning)
335 {
336 // Debug.Log($"Remaining Distance: {remainingDistance}/{movementThreshold.Start}");
337 if (ShouldStartTurning(desiredForward))
338 {
339 animator.SetBool("IsTurning", true);
340 animator.SetFloat("vel_ang", Mathf.Clamp(angleToTarget / 180f, -1f, 1f));
341 }
342 else if (remainingDistance > movementThreshold.Start)
343 {
344 animator.SetBool("IsMoving", true);
345 }
346 else
347 {
348 animator.SetFloat("vel_x", 0f, 0.2f, Time.deltaTime);
349 animator.SetFloat("vel_y", 0f, 0.2f, Time.deltaTime);
350 agent.ResetPath();
351 }
352
353 return;
354 }
355
356 //
357 // TURNING
358 //
359 if (isTurning)
360 {
361 // Debug.Log(
362 // $"Angle={angleToTarget:F1} " +
363 // $"Desired={desiredForward} " +
364 // $"Forward={animatedTransform.forward}");
365
366 float velAng = Mathf.Clamp(angleToTarget / 180f, -1f, 1f);
367 animator.SetFloat("vel_ang", velAng, 0.1f, Time.deltaTime);
368 if (facingTarget)
369 {
370 animator.SetBool("IsTurning", false);
371 animator.SetFloat("vel_ang", 0f);
372 }
373 return;
374 }
375
376 //
377 // MOVING
378 //
379 if (isMoving)
380 {
381 if (remainingDistance < (movementThreshold.Stop * animator.GetFloat("LocomotionSpeedParam")))
382 {
383 animator.SetFloat("vel_x", 0f, 0.1f, Time.deltaTime);
384 animator.SetFloat("vel_y", 0f, 0.1f, Time.deltaTime);
385 agent.ResetPath();
386 return;
387 }
388
389 // Compensate for on-the-fly extended paths
390 if (remainingDistance > cachedPathLength)
391 {
392 cachedPathLength = remainingDistance;
393 }
394
395 //
396 // ORIENTATION
397 //
398 // Keep orientation driven exclusively by steering target.
399 //
400 Vector3 steeringLocal = animatedTransform.InverseTransformDirection(desiredForward).normalized;
401 //
402 // MOVEMENT
403 //
404 // Use desiredVelocity so avoidance influences locomotion.
405 //
406 Vector3 desiredVelocity = Vector3.ProjectOnPlane(agent.desiredVelocity, surfaceUp);
407
408 Vector3 localVelocity;
409
410 if (desiredVelocity.sqrMagnitude > 0.001f)
411 {
412 localVelocity = animatedTransform.InverseTransformDirection(desiredVelocity.normalized);
413 }
414 else
415 {
416 // Fallback when the agent reports almost no velocity.
417 localVelocity = steeringLocal;
418 }
419
420 float desiredSpeed = GetSpeedAlongPath(remainingDistance, cachedPathLength);
421 desiredSpeed = Mathf.Clamp(desiredSpeed, -maximumSpeed, maximumSpeed);
422
423 animator.SetFloat("vel_x", localVelocity.x * desiredSpeed, 0.1f, Time.deltaTime);
424
425 animator.SetFloat("vel_y", localVelocity.z * desiredSpeed, 0.1f, Time.deltaTime);
426
427 if (desiredSpeed < 1f)
428 {
429 animator.SetFloat("LocomotionSpeedParam", Mathf.Lerp(0.65f, 1.0f, desiredSpeed));
430 }
431 else if (desiredSpeed > 5f)
432 {
433 animator.SetFloat("LocomotionSpeedParam", desiredSpeed / 5f);
434 }
435
436 //
437 // TURN-IN-PLACE DECISION
438 //
439 // IMPORTANT:
440 // Still use steering target angle here,
441 // NOT desiredVelocity.
442 //
443 if (remainingDistance > 0.15f &&
444 Mathf.Abs(angleToTarget) > turnWhileMovingThreshold)
445 {
446 animator.SetBool("IsMoving", false);
447 animator.SetBool("IsTurning", true);
448
449 animator.SetFloat("vel_x", 0f, 0.25f, Time.deltaTime);
450
451 animator.SetFloat("vel_y", 0f, 0.25f, Time.deltaTime);
452 cachedPathLength = remainingDistance;
453 }
454 }
455 }

487 private bool HandleOffMeshLink()
488 {
489 if (!agent.isOnOffMeshLink)
490 return false;
491
492 OffMeshLinkData linkData = agent.currentOffMeshLinkData;
493
494 int seamAreaId = NavMesh.GetAreaFromName(navSettings.PlanetSeamArea);
495 int climbAreaId = NavMesh.GetAreaFromName(navSettings.ClimbArea);
496
497 switch (linkData.offMeshLink.area)
498 {
499 case var area when area == seamAreaId:
500
501 Debug.Log($"Reached seam OffMeshLink '{linkData.offMeshLink.name}'.");
502 break;
503
504 case var area when area == climbAreaId:
505
506 Debug.Log($"Reached climb OffMeshLink '{linkData.offMeshLink.name}'.");
507 // TODO:
508 // Play climb traversal animation.
509 animator.SetBool("IsMoving", false);
510 animator.SetTrigger("ClimbTrigger");
511 break;
512
513 default:
514
515 Debug.LogWarning($"Reached OffMeshLink '{linkData.offMeshLink.name}' " + $"with unhandled area {linkData.offMeshLink.area}.");
516 break;
517 }
518
519 // Perhaps consider checking finalization condition for off mesh link traversal and then calling
520 // agent.CompleteOffMeshLink();
521
522 return true;
523 }

525 private void OnAnimatorMove()
526 {
527 animatedTransform.position = animator.rootPosition;
528 agent.nextPosition = animator.rootPosition;
529 agent.velocity = actualVelocity;
530
531 if (animator.GetBool("IsTurning"))
532 { // Root Motion Rotation
533 animatedTransform.rotation = Quaternion.FromToRotation(animatedTransform.up, surfaceUp) * animatedTransform.rotation * animator.deltaRotation;
534 }
535 else if (animator.GetBool("IsMoving"))
536 { // Procedurally controlled Rotation
537 var steeringDelta = smoothedSteeringTarget - animatedTransform.position;
538 // NOTE: we might have to project the steering target down to the tangent surfaceUp plane ( Vector3.ProjectOnPlane(..., surfaceUp); )
539 animatedTransform.rotation = Quaternion.RotateTowards(
540 animatedTransform.rotation,
541 Quaternion.LookRotation(steeringDelta.normalized, surfaceUp),
542 steeringDelta.sqrMagnitude * rotationSmoothDegrees * Time.deltaTime);
543 }
