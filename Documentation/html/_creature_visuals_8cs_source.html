244 private static void ApplyAllMaterialProperties(Renderer renderer, Color color, MaterialSettings material)
245 {
246 InitMaterialPropertyBlock();
247
248 renderer.sharedMaterial = MaterialLibrarySingleton.GetMaterial(material.type);
249 renderer.GetPropertyBlock(materialPropertyBlock);
250 materialPropertyBlock.SetColor("_BaseColor", color);
251
252 switch (material.type)
253 {
254 case MaterialOption.Fur:
255 ApplyFurProperties(material.furSettings, materialPropertyBlock);
256 break;
257 case MaterialOption.OpaqueBSDF:
258 ApplyBSDFProperties(material.bsdfSettings, materialPropertyBlock);
259 break;
260 case MaterialOption.TransparentBSDF:
261 ApplyBSDFProperties(material.bsdfSettings, materialPropertyBlock);
262 break;
263 }
264
265 renderer.SetPropertyBlock(materialPropertyBlock);
266 }
