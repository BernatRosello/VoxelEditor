242 bool TryGetVoxel(Vector2 screenPos, out Vector3Int voxel)
243 {
244 Ray ray = cam.ScreenPointToRay(screenPos);
245 bool isDeleting = EditorState.Instance.activeMesh == VoxelMeshID.VVVVVV;
246
247 var grid = gridRenderer;
248
249 // Store ray
250 debug_rayOrigin = ray.origin;
251 debug_rayDir = ray.direction;
252
253 debug_hasHit = false;
254 debug_hasAdjusted = false;
255 debug_hasBoundsHit = false;
256 debug_hasVoxel = false;
257
258 if (Physics.Raycast(ray, out RaycastHit hit))
259 {
260 debug_hasHit = true;
261
262 Vector3 point = hit.point;
263 Vector3 normal = hit.normal;
264
265 debug_hitPoint = point;
266 debug_hitNormal = normal;
267
268 Vector3 adjusted;
269
270 if (isDeleting)
271 {
272 adjusted = point - normal * 0.5f;
273 }
274 else
275 {
276 adjusted = point + normal * 0.5f;
277 }
278
279 debug_hasAdjusted = true;
280 debug_adjustedPoint = adjusted;
281
282 if (!grid.ToVoxelCoordinates(adjusted, out voxel))
283 return false;
284
285 debug_hasVoxel = true;
286 debug_voxelWorld = grid.originOffset.TransformPoint(new Vector3(voxel.x, voxel.y, voxel.z));
287
288 return true;
289 }
290 else if (grid.gridBounds.IntersectRay(ray, out float enter))
291 {
292 debug_hasBoundsHit = true;
293
294 Vector3 point = ray.GetPoint(enter);
295 debug_boundsPoint = point;
296
297 if (!grid.ToVoxelCoordinates(point, out voxel))
298 return false;
299
300 debug_hasVoxel = true;
301 debug_voxelWorld = grid.originOffset.TransformPoint(new Vector3(voxel.x, voxel.y, voxel.z));
302
303 return true;
304 }
305 voxel = default;
306 return false;
307 }
