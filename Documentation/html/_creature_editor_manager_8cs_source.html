128 private void OnClick(InputAction.CallbackContext context)
129 {
130 if (dragArea.IsDraggingCamera)
131 return;
132
133 Vector2 mousePosition = controls.Player.Point.ReadValue<Vector2>();
134
135 Ray ray = Camera.main.ScreenPointToRay(mousePosition);
136
137 if (!Physics.Raycast(ray, out var hit, 30f))
138 return;
139
140 var bodyPartCollider = hit.collider.GetComponent<BodyPartCollider>();
141
142 if (bodyPartCollider == null)
143 return;
144
145 var creature = hit.transform.GetComponentInParent<Creature>();
146 if (creature != currentCreature)
147 {
148 // Debug.Log("Changing Selected Creature");
149 SetCurrentCreature(creature);
150 }
151
152 SelectBodyPart(bodyPartCollider.BodyPart);
153 }

227 private void RebuildMaterialProperties()
228 {
229 foreach (Transform child in materialPropertiesRoot)
230 {
231 if (child.GetComponent<SliderUI>() != null)
232 Destroy(child.gameObject);
233 }
234
235 var materialSettings = CurrentVisuals.GetMaterialSettings(selectedBodyPart);
236 object settings = materialSettings.type == CreatureVisuals.MaterialOption.Fur ? materialSettings.furSettings : materialSettings.bsdfSettings;
237
238
239 var fields = settings.GetType().GetFields();
240
241 foreach (var field in fields)
242 {
243 if (field.IsLiteral)
244 continue;
245
246 if (field.FieldType != typeof(float))
247 continue;
248
249 GameObject go = Instantiate(shapeSliderPrefab, materialPropertiesRoot);
250 TMP_Text label = go.GetComponentInChildren<TMP_Text>();
251 Slider slider = go.GetComponentInChildren<Slider>();
252 label.text = field.Name;
253 RangeAttribute range = field.GetCustomAttribute<RangeAttribute>();
254
255 if (range != null)
256 {
257 if (range.min < range.max)
258 {
259 slider.minValue = range.min;
260 slider.maxValue = range.max;
261 }
262 else
263 {
264 slider.minValue = range.max;
265 slider.maxValue = range.min;
266 slider.direction = Slider.Direction.RightToLeft;
267 }
268 }
269
270 slider.SetValueWithoutNotify((float)field.GetValue(settings));
271 BindMaterialSlider(slider, field);
272 }
273 }
