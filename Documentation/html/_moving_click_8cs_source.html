113 private void UpdateOutlines()
114 {
115 ActionDriver hovered = GetHoveredCreatureDriver();
116
117 bool adding = controls.Player.Select.IsPressed();
118 bool removing = controls.Player.Remove.IsPressed();
119
120 HashSet<ActionDriver> currentlyRelevant = new(
121 selectedCreatures
122 .Select(c => c.GetComponent<ActionDriver>())
123 .Where(ad => ad != null)
124 );
125
126 if (hovered != null)
127 currentlyRelevant.Add(hovered);
128
129 // Remove outlines from creatures that are no longer relevant
130 foreach (ActionDriver creature in outlinedCreatures.Except(currentlyRelevant).ToList())
131 {
132 creature.SetOutlineNONE();
133 }
134
135 // Apply current outlines
136 foreach (ActionDriver creature in currentlyRelevant)
137 {
138 bool isHovered = creature == hovered;
139 bool isSelected = selectedCreatures.Contains(creature.GetComponent<Creature>());
140
141 // Hover modifiers have priority
142 if (removing && isHovered)
143 {
144 creature.SetOutlineRed();
145 }
146 else if (adding && isHovered)
147 {
148 creature.SetOutlineGreen();
149 }
150 else if (isSelected)
151 {
152 creature.SetOutlineBlue();
153 }
154 else
155 {
156 creature.SetOutlineNONE();
157 }
158 }
159
160 outlinedCreatures.Clear();
161
162 foreach (ActionDriver creature in currentlyRelevant)
163 {
164 outlinedCreatures.Add(creature);
165 }

260 private void CreateRequest(RaycastHit hit)
261 {
262 if (activeRequest == RequestMode.None)
263 return;
264
265 int minimumSelectionCount = minimumSelected[(int)activeRequest - 1];
266
267 if (selectedCreatures.Count < minimumSelectionCount)
268 {
269 Debug.Log($"Cannot create {activeRequest} request. Need at least {minimumSelectionCount} selected creatures.");
270 return;
271 }
272
273 switch (activeRequest)
274 {
275 case RequestMode.UserMove:
276 var moveParams = new UserMoveParams { position = hit.point };
277 foreach (var c in selectedCreatures)
278 {
279 var moveRequest = new UserMoveRequest(moveParams, c.Identity);
280 InteractionManager.CreateRequest(moveRequest);
281 }
282 break;
283
284 case RequestMode.Dance:
285 var danceParams = new CircleDanceParams { position = hit.point };
286 var danceRequest = new CircleDanceRequest(danceParams, selectedCreatures.Select(c => c.Identity));
287 InteractionManager.CreateRequest(danceRequest);
288 break;
289
290 case RequestMode.Wander:
291 var wanderParams = new WanderingParams { duration = 90, frequency = 5, frequencyVariance = 4, maxDistance = 15, minDistance = 2, moveChance = 0.5f };
292 foreach (var c in selectedCreatures)
293 {
294 var wanderRequest = new WanderingRequest(wanderParams, c.Identity);
295 InteractionManager.CreateRequest(wanderRequest);
296 }
297 break;
298
299 case RequestMode.Idle:
300
301 IdleParams idleParams = new()
302 {
303 duration = 30,
304 avgEmotesPerMinute = m_avgEmotesPerMinute
305 };
306 foreach (var c in selectedCreatures)
307 {
308 InteractionManager.CreateRequest(new IdleInteractionRequest(idleParams, c.Identity));
309 }
310 break;
311
312 case RequestMode.Talk:
313
314 TalkParams talkParams = new()
315 {
316 duration = 20,
317 };
318 InteractionManager.CreateRequest(new TalkInteractionRequest(talkParams, selectedCreatures.Select(c => c.Identity)));
319 break;
320
321 }
322
323 // Request successfully created -> clear selection
324 selectedCreatures.Clear();
325 }
